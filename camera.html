from flask import Flask, render_template, redirect from voice import speak import subprocess app = Flask(__name__) # Login Page @app.route("/") def login(): return render_template("login.html") # Register Page @app.route("/register") def register(): return render_template("register.html") # Dashboard @app.route("/dashboard") def dashboard(): return render_template("dashboard.html") # Start Camera @app.route("/camera") def camera(): subprocess.Popen(["python", "camera.py"]) return "Camera Started" # Voice Assistant @app.route("/voice") def voice(): speak("Welcome to Object Detection For Blind") return "Voice Assistant Started" # SOS Emergency @app.route("/sos") def sos(): speak("Emergency Alert Activated") return "Emergency Alert Activated" # Detection History @app.route("/history") def history(): return render_template("history.html") # Settings @app.route("/settings") def settings(): return render_template("settings.html") # GPS Navigation @app.route("/gps") def gps(): speak("GPS Navigation Started") return render_template("gps.html") # Logout @app.route("/logout") def logout(): return redirect("/") if __name__ == "__main__": app.run(debug=True)
